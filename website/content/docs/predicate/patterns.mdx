## E-Commerce Filtering

### Product Search & Discovery

```typescript
import * as R from 'remeda'
import { where, between, gt, gte, eq, oneOf, by, and, or } from '@ontologyio/receta/predicate'

interface Product {
  id: string
  name: string
  price: number
  category: string
  rating: number
  reviewCount: number
  inStock: boolean
  quantity: number
  salesLast30Days: number
  tags: string[]
  brand: string
  discount: number  // 0.0 to 1.0
}

// Price slider: $50-200
const priceRange = where({
  price: between(50, 200)
})

// Quality products (rating >= 4, enough reviews)
const qualityProduct = where({
  rating: gte(4.0),
  reviewCount: gte(10)
})

// Available products
const available = where({
  inStock: eq(true)
})

// Electronics category
const electronics = where({
  category: eq('electronics')
})

// On sale (discount > 0)
const onSale = where({
  discount: gt(0)
})

// Combine for "Best Electronics Deals"
const bestElectronicsDeals = and<Product>(
  electronics,
  qualityProduct,
  available,
  onSale,
  priceRange
)

const deals = R.filter(products, bestElectronicsDeals)
```

### Dynamic Search Filters

```typescript
interface SearchFilters {
  category?: string
  minPrice?: number
  maxPrice?: number
  minRating?: number
  brands?: string[]
  onlyInStock?: boolean
  tags?: string[]
}

// A filter the user left out becomes always(), which lets every product through
const buildProductFilter = ({
  category,
  minPrice = 0,
  maxPrice = Infinity,
  minRating = 0,
  brands = [],
  onlyInStock = false,
  tags = []
}: SearchFilters): Predicate<Product> =>
  and<Product>(
    category ? where({ category: eq(category) }) : always(),
    where({ price: between(minPrice, maxPrice), rating: gte(minRating) }),
    isEmpty(brands) ? always() : where({ brand: oneOf(brands) }),
    onlyInStock ? where({ inStock: eq(true) }) : always(),
    isEmpty(tags)
      ? always()
      : where({ tags: (productTags: string[]) => tags.some(oneOf(productTags)) })
  )

// Usage
const searchResults = R.filter(
  products,
  buildProductFilter({
    category: 'electronics',
    minPrice: 100,
    maxPrice: 500,
    minRating: 4.0,
    brands: ['Apple', 'Samsung'],
    onlyInStock: true
  })
)
```

### Inventory Management

```typescript
// Low stock alert (< 10 units)
const lowStock = where({
  quantity: lt(10),
  inStock: eq(true)

// Out of stock but popular (needs reorder)
```

## User Permissions & Access Control

### Role-Based Access Control

```typescript
interface User {
  id: string
  role: 'user' | 'moderator' | 'admin' | 'owner'
  verified: boolean
  accountAgeDays: number
  reputation: number
  permissions: string[]
  flags: string[]
  tier: 'free' | 'pro' | 'enterprise'
}

// Basic roles
const isAdmin = where({ role: eq('admin') })
const isModerator = where({ role: eq('moderator') })
const isOwner = where({ role: eq('owner') })

// Staff members (admins or moderators)
const isStaff = where({
  role: oneOf(['admin', 'moderator', 'owner'])
})

// Verified users
const isVerified = where({
  verified: eq(true)
})

// Veteran users (account > 1 year)
const isVeteran = where({
  accountAgeDays: gte(365)
})

// High reputation users
const highReputation = where({
  reputation: gte(1000)
})

// Premium users
const isPremium = where({
  tier: oneOf(['pro', 'enterprise'])
})
```

### Complex Permission Logic

```typescript
// Can moderate content
const canModerate = or<User>(

// Can edit any post
const canEditAnyPost = or<User>(
  isAdmin,
  and<User>(isModerator, isVerified, highReputation)
)

// Can access admin panel
const canAccessAdminPanel = or<User>(
  isAdmin,
  and<User>(isModerator, isVeteran, isVerified)
)

// Can create premium content
const canCreatePremiumContent = or<User>(
  isPremium,
  isStaff
)

// Can invite users
const canInviteUsers = or<User>(
  isStaff,
  and<User>(isVerified, highReputation, isVeteran)
)

// Trusted user (can skip moderation)
const isTrusted = or<User>(
  isStaff,
  and<User>(
    isVerified,
    highReputation,
    where({ accountAgeDays: gte(180) }),
    where({ flags: isEmpty })
  )
)
```

### Resource-Level Permissions

```typescript
interface Resource {
  id: string
  ownerId: string
  public: boolean
  collaborators: string[]
}

const canView = (user: User, resource: Resource): boolean =>
  or<User>(
    // Owner can always view
    where({ id: eq(resource.ownerId) }),
    // Public resources
    () => resource.public,
    // Collaborators
    where({ id: oneOf(resource.collaborators) }),
    // Admins can view everything
    isAdmin
  )(user)

const canEdit = (user: User, resource: Resource): boolean =>
  or<User>(
    // Owner can edit
    where({ id: eq(resource.ownerId) }),
    // Admins can edit
    isAdmin,
    // Verified collaborators can edit
    and<User>(
      where({ id: oneOf(resource.collaborators) }),
      isVerified
    )
  )(user)

const canDelete = (user: User, resource: Resource): boolean =>
  or<User>(
    where({ id: eq(resource.ownerId) }),
    isAdmin
  )(user)
```

## API Filtering (GitHub/Stripe Style)

### GitHub Issues Filtering

```typescript
interface Issue {
  id: number
  number: number
  state: 'open' | 'closed'
  title: string
  body: string
  labels: string[]
  assignee: string | null
  milestone: string | null
  comments: number
  createdAt: Date
  updatedAt: Date
  author: string
}

// Open issues
const open = where({ state: eq('open') })

// Assigned issues
const assigned = where({ assignee: isDefined })

// Unassigned issues
const unassigned = where({ assignee: isNull })

// Has specific label
const hasLabel = (label: string) =>
  where({
    labels: (labels: string[]) => labels.includes(label)
  })

// Bugs
const isBug = hasLabel('bug')

// Features
const isFeature = hasLabel('feature')

// Critical issues
const isCritical = hasLabel('critical')

// Needs triage (open, no labels, no assignee)
const needsTriage = and<Issue>(
  open,
  unassigned,
  where({ labels: isEmpty })
)

// Stale issues (open, no activity in 90 days)
const isStale = and<Issue>(
  open,
  where({
    updatedAt: (date: Date) => {
      const ninetyDaysAgo = Date.now() - 90 * 24 * 60 * 60 * 1000
      return date.getTime() < ninetyDaysAgo
    }
  })
)

// High priority bugs (open, bug label, critical or many comments)
const highPriorityBugs = and<Issue>(
  open,
  isBug,
  or<Issue>(
    isCritical,
    where({ comments: gte(10) })
  )
)
```

### Stripe Charges Filtering

```typescript
interface Charge {
  id: string
  amount: number  // cents
  currency: string
  status: 'succeeded' | 'pending' | 'failed'
  paid: boolean
  refunded: boolean
  disputed: boolean
  customerId: string
  created: number  // unix timestamp
}

// Successful charges
const succeeded = where({
  status: eq('succeeded'),
  paid: eq(true)
})

// Failed charges
const failed = where({ status: eq('failed') })

// Large charges (> $100)
const largeCharges = where({
  amount: gt(10000)  // $100.00 in cents
})

// Recent charges (last 30 days)
const recent = where({
  created: (timestamp: number) => {
    const thirtyDaysAgo = Date.now() / 1000 - 30 * 24 * 60 * 60
    return timestamp > thirtyDaysAgo
  }
})

// Refunded charges
const refunded = where({
  refunded: eq(true)
})

// Disputed charges
const disputed = and<Charge>(
  succeeded,
  where({ disputed: eq(true) })
)
```

## Form Validation

### Signup Form Validation

```typescript
import { and, by, eq, gte, matches, satisfies, where } from '@ontologyio/receta/predicate'
import { fromPredicate, validate } from '@ontologyio/receta/validation'

interface SignupForm {
  email: string
  password: string
  confirmPassword: string
  age: number
  agreeToTerms: boolean
  newsletter: boolean
}

// Email validation
const isValidEmail = matches(/^[^\s@]+@[^\s@]+\.[^\s@]+$/)

// Password requirements
const isStrongPassword = and(
  by((s: string) => s.length, gte(8)),
  matches(/[A-Z]/),      // Has uppercase
  matches(/[a-z]/),      // Has lowercase
  matches(/[0-9]/),      // Has number
  matches(/[!@#$%^&*]/)  // Has special char
)

// Age validation
const isAdult = gte(18)

const passwordsMatch = satisfies((form: SignupForm) =>
  form.password === form.confirmPassword
)

// Validate entire form. Every failed check adds its message.
const validateSignup = validate<SignupForm, string>([
  fromPredicate(where<SignupForm>({ email: isValidEmail }), 'Invalid email address'),
  fromPredicate(
    where<SignupForm>({ password: isStrongPassword }),
    'Password must be 8+ chars with uppercase, lowercase, number, and special char'
  ),
  fromPredicate(passwordsMatch, 'Passwords do not match'),
  fromPredicate(where<SignupForm>({ age: isAdult }), 'Must be 18 or older'),
  fromPredicate(where<SignupForm>({ agreeToTerms: eq(true) }), 'Must agree to terms and conditions')
])
```

### Multi-Step Form Validation

```typescript
import { between, by, isNotEmpty, matches, oneOf, where } from '@ontologyio/receta/predicate'

interface Step1Data {
  firstName: string
  lastName: string
  email: string
}

interface Step2Data {
  address: string
  city: string
  zipCode: string
  country: string
}

interface Step3Data {
  cardNumber: string
  cvv: string
  expiryDate: string
}

const validateStep1 = where({
  firstName: by((s: string) => s.length, between(2, 50)),
  lastName: by((s: string) => s.length, between(2, 50)),
  email: matches(/^[^\s@]+@[^\s@]+\.[^\s@]+$/)
})

const validateStep2 = where({
  address: isNotEmpty,
  city: isNotEmpty,
  zipCode: matches(/^\d{5}(-\d{4})?$/),
  country: oneOf(['US', 'CA', 'UK', 'AU'])
})

const validateStep3 = where({
  cardNumber: matches(/^\d{16}$/),
  cvv: matches(/^\d{3,4}$/),
  expiryDate: matches(/^\d{2}\/\d{2}$/)
})
```

## Search & Query Functionality

### Full-Text Search with Filters

```typescript
interface Article {
  id: string
  title: string
  content: string
  author: string
  tags: string[]
  publishedAt: Date
  viewCount: number
  featured: boolean
}

const searchArticles = (
  articles: Article[],
  query: string,
  filters: {
    author?: string
    tags?: string[]
    onlyFeatured?: boolean
    minViews?: number
  } = {}
) => {
  const { author, tags = [], onlyFeatured = false, minViews = 0 } = filters
  const searchTerm = query.toLowerCase()

  // Text search predicate
  const matchesSearch = or(
    by((a: Article) => a.title.toLowerCase(), includes(searchTerm)),
    by((a: Article) => a.content.toLowerCase(), includes(searchTerm))
  )

  // Combine the search with whichever filters were given
  return R.filter(
    articles,
    and<Article>(
      matchesSearch,
      author ? where({ author: eq(author) }) : always(),
      isEmpty(tags)
        ? always()
        : where({ tags: (articleTags: string[]) => tags.some(oneOf(articleTags)) }),
      onlyFeatured ? where({ featured: eq(true) }) : always(),
      where({ viewCount: gte(minViews) })
    )
  )
}
```

### Faceted Search

```typescript
// Count results per facet
const getFacetCounts = (products: Product[]) => {
  const categories = R.countBy(products, (p) => p.category)
  const brands = R.countBy(products, (p) => p.brand)

  const priceRanges = {
    under50: R.filter(products, where({ price: lt(50) })).length,
    '50to100': R.filter(products, where({ price: between(50, 100) })).length,
    '100to200': R.filter(products, where({ price: between(100, 200) })).length,
    over200: R.filter(products, where({ price: gt(200) })).length
  }

  return { categories, brands, priceRanges }
}
```

## Batch Processing & Data Pipelines

### Data Quality Checks

```typescript
interface DataRecord {
  id: string
  value: number
  timestamp: Date
  source: string
  validated: boolean
}

// Valid records (non-negative value, recent, validated)
const isValid = where({
  value: gte(0),
  timestamp: (date: Date) => {
    const oneDayAgo = Date.now() - 24 * 60 * 60 * 1000
    return date.getTime() > oneDayAgo
  },
  validated: eq(true)
})

// Suspicious records (extreme values)
const isSuspicious = or<DataRecord>(
  where({ value: lt(0) }),
  where({ value: gt(1000000) })
)

// Partition data
const [valid, invalid] = R.partition(records, isValid)
const suspicious = R.filter(records, isSuspicious)
```

### ETL Pipeline

```typescript
const processRecords = (rawData: unknown[]) => {
  // Step 1: Filter valid objects
  const objects = R.filter(rawData, isObject)

  // Step 2: Parse and validate
  const parsed: ParsedRecord[] = R.pipe(
    objects,
    R.map(parseRecord),
    R.filter(isDefined)
  )

  // Step 3: Apply business rules
  const validRecords = R.filter(
    parsed,
    where({
      amount: and(gte(0), lt(1000000)),
      status: oneOf(['active', 'pending']),
      timestamp: (d: Date) => !isNaN(d.getTime())
    })
  )

  return validRecords
}
```

## Content Moderation

```typescript
interface Post {
  id: string
  content: string
  authorId: string
  flagCount: number
  reportedReasons: string[]
  createdAt: Date
}

// Needs review (multiple flags or serious reports)
const needsReview = or<Post>(
  where({ flagCount: gte(3) }),
  where({
    reportedReasons: (reasons: string[]) =>
      reasons.some(oneOf(['spam', 'hate', 'violence']))
  })
)

// Auto-approve (trusted author, no flags)
const autoApprove = and<Post>(
  where({ flagCount: eq(0) }),
  where({ authorId: oneOf(trustedAuthors) })
)

// Needs manual review (somewhere in between)
const manualReview = and(
  not(autoApprove),
  not(needsReview)
)
```
